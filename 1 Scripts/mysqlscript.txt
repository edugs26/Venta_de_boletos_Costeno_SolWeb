# Triggers #

-- Cambiar el delimitador para definir los triggers
DELIMITER $$

-- Trigger 1: Actualizar los asientos a 'OCUPADO' cuando el viaje se desactiva
CREATE TRIGGER actualizar_asientos_inactivos
AFTER UPDATE ON viaje
FOR EACH ROW
BEGIN
  -- Verificar si el estado del viaje cambió a 'inactivo'
  IF NEW.estado_viaje = 'inactivo' AND OLD.estado_viaje <> 'inactivo' THEN
    -- Actualizar los asientos del bus asociado, estableciendo id_estado_asiento = 2 ('OCUPADO')
    UPDATE asiento
    SET id_estado_asiento = 2  -- '2' corresponde al estado 'OCUPADO' en la tabla estado_asiento
    WHERE id_bus = NEW.id_bus;
  END IF;
END$$

-- Trigger 2: Establecer los asientos como 'LIBRES' al insertar un nuevo viaje activo
CREATE TRIGGER activar_asientos_libres_insert
AFTER INSERT ON viaje
FOR EACH ROW
BEGIN
  -- Verificar si el nuevo viaje tiene estado 'activo'
  IF NEW.estado_viaje = 'activo' THEN
    -- Actualizar los asientos del bus asociado, estableciendo id_estado_asiento = 1 ('LIBRE')
    UPDATE asiento
    SET id_estado_asiento = 1  -- '1' corresponde al estado 'LIBRE' en la tabla estado_asiento
    WHERE id_bus = NEW.id_bus;
  END IF;
END$$

-- Paso previo para Trigger 3: Añadir el campo fecha_activacion a la tabla viaje

-- Trigger 3: Establecer los asientos como 'LIBRES' al activar un viaje existente por primera vez
CREATE TRIGGER activar_asientos_libres_update
AFTER UPDATE ON viaje
FOR EACH ROW
BEGIN
  -- Verificar si el estado del viaje cambió a 'activo' y aún no ha sido activado antes
  IF NEW.estado_viaje = 'activo' AND OLD.estado_viaje <> 'activo' AND NEW.fecha_activacion IS NULL THEN
    -- Actualizar los asientos del bus asociado, estableciendo id_estado_asiento = 1 ('LIBRE')
    UPDATE asiento
    SET id_estado_asiento = 1  -- '1' corresponde al estado 'LIBRE' en la tabla estado_asiento
    WHERE id_bus = NEW.id_bus;
    
    -- Establecer la fecha de activación para indicar que el viaje ya ha sido activado
    UPDATE viaje
    SET fecha_activacion = NOW()
    WHERE id_viaje = NEW.id_viaje;
  END IF;
END$$
